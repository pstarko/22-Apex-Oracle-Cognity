--Krok 1. Utwórz użytkownika/schemat aplikacyjny

CREATE USER apex_app
IDENTIFIED BY "Oracle123$"
DEFAULT TABLESPACE users
TEMPORARY TABLESPACE temp
QUOTA UNLIMITED ON users;

--Następnie nadaj niezbędne uprawnienia do pracy z własnymi obiektami:
GRANT CREATE SESSION TO apex_app; -- jak do odczytu to uprawnienie jest minimalne
GRANT CREATE TABLE TO apex_app;
GRANT CREATE SEQUENCE TO apex_app;
GRANT CREATE PROCEDURE TO apex_app;
GRANT CREATE TRIGGER TO apex_app;
GRANT CREATE VIEW TO apex_app;

--Krok 2. Utwórz przykładową tabelę i dane
--Połącz się teraz jako nowy użytkownik APEX_APP.

CREATE TABLE customers (
    customer_id   NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    first_name    VARCHAR2(100) NOT NULL,
    last_name     VARCHAR2(100) NOT NULL,
    email         VARCHAR2(255),
    created_at    TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL
);

INSERT INTO customers (first_name, last_name, email)
VALUES ('Anna', 'Kowalska', 'anna.kowalska@example.com');

INSERT INTO customers (first_name, last_name, email)
VALUES ('Jan', 'Nowak', 'jan.nowak@example.com');

COMMIT;

SELECT customer_id, first_name, last_name, email, created_at
FROM customers;
