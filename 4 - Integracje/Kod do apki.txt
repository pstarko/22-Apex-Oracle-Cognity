declare  
    l_url          varchar2(4000);  
    l_response     clob;  
    l_latitude     number := :P17_LATITUDE;  
    l_longitude    number := :P17_LONGITUDE;  
    l_temperature  number;  
    l_wind_speed   number;  
    l_weather_code number;  
begin  
    if l_latitude is null or l_latitude not between -90 and 90 then  
        raise_application_error(  
            -20001,  
            'Latitude musi znajdować się w zakresie od -90 do 90.'  
        );  
    end if;  
  
    if l_longitude is null or l_longitude not between -180 and 180 then  
        raise_application_error(  
            -20002,  
            'Longitude musi znajdować się w zakresie od -180 do 180.'  
        );  
    end if;  
  
    l_url :=  
        'https://api.open-meteo.com/v1/forecast' ||  
        '?latitude='  || to_char(l_latitude,  'FM990D999999', 'NLS_NUMERIC_CHARACTERS=.,') ||  
        '&longitude=' || to_char(l_longitude, 'FM990D999999', 'NLS_NUMERIC_CHARACTERS=.,') ||  
        '&current=temperature_2m,wind_speed_10m,weather_code' ||  
        '&timezone=auto';  
  
    l_response := apex_web_service.make_rest_request(  
        p_url         => l_url,  
        p_http_method => 'GET'  
    );  
  
    apex_json.parse(l_response);  
  
    l_temperature := apex_json.get_number(  
        p_path => 'current.temperature_2m'  
    );  
  
    l_wind_speed := apex_json.get_number(  
        p_path => 'current.wind_speed_10m'  
    );  
  
  
    :P17_TEMP  := to_char(l_temperature) || ' °C';  
    :P17_WIND   := to_char(l_wind_speed) || ' km/h';  

  
exception  
    when others then  
        :P17_TEMP   := null;  
        :P17_WIND    := null;    
        raise;  
end;
